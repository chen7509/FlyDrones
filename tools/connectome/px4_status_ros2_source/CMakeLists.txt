cmake_minimum_required(VERSION 3.16)
project(flydrones_px4_status_ros2_source)

find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(OpenSSL REQUIRED)

get_filename_component(PX4_MSGS_PREFIX "${px4_msgs_DIR}/../../.." ABSOLUTE)
set(PX4_STATUS_MSG "${PX4_MSGS_PREFIX}/share/px4_msgs/msg/EstimatorStatus.msg")
if(NOT EXISTS "${PX4_STATUS_MSG}")
  message(FATAL_ERROR "Installed px4_msgs EstimatorStatus.msg missing: ${PX4_STATUS_MSG}")
endif()
file(SHA256 "${PX4_STATUS_MSG}" PX4_STATUS_MSG_SHA256)
if(NOT PX4_STATUS_MSG_SHA256 STREQUAL "44cba4424c224753b8249729e9bf7e530d0b34e04d5299fcf94f2677c8f50bd5")
  message(FATAL_ERROR "Installed px4_msgs EstimatorStatus.msg does not match pinned PX4")
endif()

add_executable(px4_estimator_status_source px4_estimator_status_source.cpp)
ament_target_dependencies(px4_estimator_status_source ament_index_cpp rclcpp px4_msgs)
target_link_libraries(px4_estimator_status_source OpenSSL::Crypto)
target_compile_features(px4_estimator_status_source PRIVATE cxx_std_17)
target_compile_options(px4_estimator_status_source PRIVATE -Wall -Wextra -Werror)
install(TARGETS px4_estimator_status_source DESTINATION lib/${PROJECT_NAME})
ament_package()
