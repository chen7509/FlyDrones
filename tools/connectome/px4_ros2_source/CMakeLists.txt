cmake_minimum_required(VERSION 3.16)
project(flydrones_px4_ros2_source)

find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(OpenSSL REQUIRED)

get_filename_component(PX4_MSGS_PREFIX "${px4_msgs_DIR}/../../.." ABSOLUTE)
set(PX4_ODOMETRY_MSG "${PX4_MSGS_PREFIX}/share/px4_msgs/msg/VehicleOdometry.msg")
if(NOT EXISTS "${PX4_ODOMETRY_MSG}")
  message(FATAL_ERROR "Installed px4_msgs VehicleOdometry.msg missing: ${PX4_ODOMETRY_MSG}")
endif()
file(SHA256 "${PX4_ODOMETRY_MSG}" PX4_ODOMETRY_MSG_SHA256)
if(NOT PX4_ODOMETRY_MSG_SHA256 STREQUAL "a528b3d0b4c1a9083a71b32c367bad71a900e959efd82b9d03a9eb0e99fe6017")
  message(FATAL_ERROR "Installed px4_msgs VehicleOdometry.msg does not match pinned PX4")
endif()

add_executable(px4_odometry_source px4_odometry_source.cpp)
ament_target_dependencies(px4_odometry_source ament_index_cpp rclcpp px4_msgs)
target_link_libraries(px4_odometry_source OpenSSL::Crypto)
target_compile_features(px4_odometry_source PRIVATE cxx_std_17)
target_compile_options(px4_odometry_source PRIVATE -Wall -Wextra -Werror)
install(TARGETS px4_odometry_source DESTINATION lib/${PROJECT_NAME})
ament_package()
